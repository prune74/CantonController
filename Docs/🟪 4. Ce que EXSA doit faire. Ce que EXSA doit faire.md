HardwareSerial* ServoUart::m_uart = nullptr;

    m_uart = &uart;

uint8_t ServoUart::checksum(const uint8_t *data, uint8_t len)
{
    uint8_t cs = 0;
    for (uint8_t i = 0; i < len; i++)
        cs ^= data[i];
    return cs;
}

void ServoUart::sendServoTest(uint8_t idServo)
{
    if (!m_uart) return;

    uint8_t frame[3];
    frame[0] = 0xF0;
    frame[1] = idServo;
    frame[2] = checksum(frame, 2);

    m_uart->write(frame, 3);
}

void ServoUart::sendServoConfig(uint8_t idServo, uint16_t posD, uint16_t posV, uint8_t speed)
